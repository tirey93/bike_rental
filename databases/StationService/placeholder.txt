Table ExternalBikes {
  Id integer
  Color text
  ExternalId text
  Model text
}

Table BikesAtStation {
  Id integer [primary key]
  BikeId integer
  StationId integer
}

Table Stations {
  Id integer [primary key]
  ExternalId integer
  Capacity integer
  Code integer
  Location integer
}


Ref FK_ExternalBikeAtStations_Stations_StationId: BikesAtStation.StationId > Stations.Id // many-to-one
Ref FK_ExternalBikeAtStations_Bike_StationId: BikesAtStation.BikeId > ExternalBikes.Id // many-to-one
